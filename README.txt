CONTROLS:

interaction - 
	LMB: pick up
	RMB: use
	Scroll: select held item
	Q & E: rotate held item
	G: drop held item
on ground - 
	Space: jump
	hold Space: hover
while hovering - 
	Space: increase height
	Ctrl: decrease height
	double tap Space: begin flight
while flying - 
	W - pitch down
	S - pitch up
	Space - strong flap
	hold Space - constant flap
	Ctrl - slow down
hold Alt  -  disable auto camera
hold Tab -  show nest markers
F     -  toggle HUD
